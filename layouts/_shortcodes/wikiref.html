{{- /* Renders an inline reference to a wiki entry: styled like an (dotted underline), with a themed popup on hover/focus showing the entry's summary and a "[read more]" link through to it. @param {string} [slug] The wiki entry's slug (content/wiki/.md). Optional if the slug is the first and only positional argument. @returns {template.HTML} @example {{< wikiref "great-lent" >}}Great Lent{{< /wikiref >}} @example {{< wikiref slug="great-lent" />}} */}} {{- $slug := or (.Get "slug") (.Get 0) -}} {{- with $slug -}} {{- $entry := site.GetPage (printf "/wiki/%s" .) -}} {{- if not $entry -}} {{- errorf "The %q shortcode found no wiki entry at content/wiki/%s.md. See %s" $.Name $slug $.Position -}} {{- else -}} {{- $text := or $.Inner $entry.Title -}} {{- $summary := $entry.Summary | plainify | strings.TrimSpace -}} {{ $text }}{{ $summary }} [read more] {{- end -}} {{- else -}} {{- errorf "The %q shortcode requires a slug argument. See %s" .Name .Position -}} {{- end -}}
